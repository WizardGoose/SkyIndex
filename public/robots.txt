# Everything here is crawlable, which is not the same as wanting to be indexed.
# Every page carries <meta name="robots" content="noindex, nofollow">, and a
# crawler has to be allowed to fetch a page before it can read that tag.
# Disallowing here would do the opposite of the intent: the address stays
# indexable from any link pointing at it, and the instruction never gets read.
#
# This root-hosted site serves robots.txt at the crawler's expected path.
#
# No Sitemap line, because there is no sitemap.

User-agent: *
Allow: /
